vint16m1_t vpu_exp_quantized_q88(vint16m1_t x_q8, size_t vl) {
  /*
   * Quantized exponential approximation for one Q8.8 input vector.
   * Input:  x_q8  — Q8.8 format (vint16m1_t)
   * Output: exp(x) approximated in Q0.8 format (vint16m1_t, range 0~255)
   */

    // 1. log2(e) ≈ 1.5 → x * log2(e) ≈ x + x/2
    vint16m1_t int_frac = __riscv_vadd_vv_i16m1(x_q8, __riscv_vsra_vx_i16m1(x_q8, 1, vl), vl); // Q8.8
    // print result
    // int16_t int_frac_arr[vl];
    // __riscv_vse16_v_i16m1(int_frac_arr, int_frac, vl);
    // for (int i = 0; i < vl; i++) {
    //     printf("int_frac_arr[%d]: %d\n", i, int_frac_arr[i]);
    // }

    // 2. split into integer and fractional parts
    vint16m1_t integer_part = __riscv_vsra_vx_i16m1(int_frac, 8, vl);             // signed integer part
    vint16m1_t frac_part    = __riscv_vand_vx_i16m1(int_frac, 0x00FF, vl); // fractional part (0~255)
    // print result
    // int16_t integer_part_arr[vl];
    // int16_t frac_part_arr[vl];
    // __riscv_vse16_v_i16m1(integer_part_arr, integer_part, vl);
    // __riscv_vse16_v_i16m1(frac_part_arr, frac_part, vl);

    // for (int i = 0; i < vl; i++) {
    //     printf("part_arr[%d]: %d, %d\n", i, integer_part_arr[i], frac_part_arr[i]);
    // }

    // 3. compute 2^integer_part
    // (1 << 8) means 1.0 in Q0.8 format
    vuint16m1_t shift_amt = __riscv_vreinterpret_v_i16m1_u16m1(__riscv_vneg_v_i16m1(integer_part, vl)); // abs(negative)
    vint16m1_t exp_int = __riscv_vsra_vv_i16m1(__riscv_vmv_v_x_i16m1(1 << 8, vl), shift_amt, vl);

    // 4. approximate 2^fractional ≈ 1 + frac/2
    vint16m1_t exp_frac = __riscv_vadd_vv_i16m1(__riscv_vmv_v_x_i16m1(1 << 8, vl), __riscv_vsra_vx_i16m1(frac_part, 1, vl), vl); // still Q0.8
    // print result
    // int16_t exp_int_arr[vl];
    // int16_t exp_frac_arr[vl];
    // __riscv_vse16_v_i16m1(exp_int_arr, exp_int, vl);
    // __riscv_vse16_v_i16m1(exp_frac_arr, exp_frac, vl);

    // for (int i = 0; i < vl; i++) {
    //     printf("exp_part_arr[%d]: %d, %d\n", i, exp_int_arr[i], exp_frac_arr[i]);
    // }

    // 5. combine integer and fractional parts → Q0.8 output
    vint16m1_t exp_out_w = __riscv_vmul_vv_i16m1(exp_int, exp_frac, vl);
    // vint32m2_t exp_out_w = __riscv_vwmul_vv_i32m2(exp_int, exp_frac, vl);
    // print result
    // int16_t exp_out_w_arr[vl];
    // __riscv_vse16_v_i16m1(exp_out_w_arr, exp_out_w, vl);
    // for (int i = 0; i < vl; i++) {
    //     printf("exp_out_w_arr[%d]: %04x\n", i, exp_out_w_arr[i]);
    // }

    vint16m1_t exp_out = __riscv_vsra_vx_i16m1(exp_out_w, 8, vl);


    return exp_out;
}

void vpu_softmax_q88(int vl, int16_t* INPUT_VECTOR, int16_t* SM_RESULT) {
  // 初始化最大值為最小可能的16位整數
  int16_t maximum = INT16_MIN;

  size_t vlmax;
  vint16m1_t v_data, v_max, v_res;
  vint16m1_t v_zero = __riscv_vmv_v_x_i16m1(0, 1); // 初始值(可省略)
  

  // reduction maximum
  for (int i = 0; i < vl; i += vlmax) {
      vlmax = __riscv_vsetvl_e16m1(vl - i);
      // 載入資料
      v_data = __riscv_vle16_v_i16m1(&INPUT_VECTOR[i], vlmax);

      // 歸約取最大值 (若只有vredsum，可替換成sum示範)
      v_max = __riscv_vredmax_vs_i16m1_i16m1(v_data, v_zero, vlmax);

      // 取出 scalar 結果
      int16_t temp = __riscv_vmv_x_s_i16m1_i16(v_max);

      if (temp > maximum) {
          maximum = temp;
      }
  }

  // printf("Maximum value: 0x%04x\n", maximum);


  // vint16m1_t __riscv_vmv_v_x_i16m1 (int16_t src, size_t vl);
  // vint16m1_t __riscv_vredmax_vs_i16m1_i16m1 (vint16m1_t vector, vint16m1_t scalar, size_t vl);
  // printf("Maximum value: 0x%04x\n", (uint16_t)max_val);


  // element-wise subtraction
  int16_t sub_max[vl];
  for (int i = 0; i < vl; i += vlmax) {
      vlmax = __riscv_vsetvl_e16m1(vl - i);
      v_data = __riscv_vle16_v_i16m1(&INPUT_VECTOR[i], vlmax);

      // result = maximum - v_data
      v_res = __riscv_vsub_vx_i16m1(v_data, maximum, vlmax);

      // store back to sub_max[]
      __riscv_vse16_v_i16m1(&sub_max[i], v_res, vlmax);
  }

  // print result
  // for (int i = 0; i < vl; i++) {
  //     printf("sub_max[%d]: 0x%04x\n", i, sub_max[i]);
  // }
  // vint16m1_t __riscv_vsub_vx_i16m1 (vint16m1_t op1, int16_t op2, size_t vl);


  // element-wise exponentiation
  int16_t v_exp[vl];

  v_data = vpu_exp_quantized_q88(v_res, vl);
  
  __riscv_vse16_v_i16m1(&v_exp, v_data, vl);

  // print result
  // for (int i = 0; i < vl; i++) {
  //     printf("v_exp[%d]: 0x%04x\n", i, v_exp[i]);
  // }

  // reduction summation
  vint16m1_t summation =  __riscv_vredsum_vs_i16m1_i16m1 (v_data, v_zero, vl);

  int16_t temp = __riscv_vmv_x_s_i16m1_i16(summation);
  temp = temp + 0x00100; // this is because vpu exp cannot support 16b*16b = 32b operation !!
  // printf("Exponential sum: 0x%04x\n", (uint16_t)temp);

  // element-wise reciprocal
  int16_t reciprocal = (1 << 16) / temp;
  // printf("Reciprocal: 0x%04x\n", (int16_t)reciprocal);

  vint16m1_t sm_result_temp = __riscv_vmul_vx_i16m1(v_data, reciprocal, vl);
  vint16m1_t sm_result = __riscv_vsra_vx_i16m1(sm_result_temp, 8, vl);

  int16_t v_sm[vl];
  __riscv_vse16_v_i16m1(&v_sm, sm_result, vl);

  // // print result
  // for (int i = 0; i < vl; i++) {
  //     printf("v_sm[%d]: 0x%04x\n", i, v_sm[i]);
  // }
}
